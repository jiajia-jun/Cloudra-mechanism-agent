ChatResponse = Union[ChatCompletion, Stream[ChatCompletionChunk]]

class Agent(BaseModel):
    '''agent父类，用于定义各类agent属性：
    1.唯一标识符（id）
    2.agent名称(name)
    3.agent类型(type)
    4.agent使用模型（model）
    5.agent使用模型url（llm_site）
    6.agent使用模型key（_api_key）
    7.agent使用模型温度（temperature）
    8.agent使用模型最大token（max_tokens）
    9.agent使用模型top_p（top_p）
    10.agent使用模型频率惩罚（frequency_penalty）
    11.agent使用模型存在惩罚（presence_penalty）
    12.agent身份prompt（prompt）
    13.agent上下文（context）
    14.agent输出方式（is_stream）

    该类的其他方法:
    权限：has_write_permission，has_read_permission，script_run_permission，destructive_permission(均为布尔值输出)
    llm_call方法：调用llm，输出ChatResponse或ChatStream
    后续可能加入的方法：调用本地专用模型，不能通过openai接口，因此待定，可以上github上看看有没有开源的逆合成分析模型等
    '''
    #唯一标识符
    id: str

    #agent名称
    name: str

    #agent类型：coordinator,researcher,...other
    type: str

    #agent使用模型
    model: str

    #agent使用模型url
    llm_site: str

    #agent使用模型key
    _api_key: str

    #agent使用模型温度
    temperature: float = 0.7

    #agent使用模型最大token
    max_tokens: int

    #agent使用模型top_p
    top_p: float = 1.0

    #agent使用模型频率惩罚
    frequency_penalty: float = 0.0

    #agent使用模型存在惩罚
    presence_penalty: float = 0.0

    #agent身份prompt，如：“你是一个多agent协作体系的协调器......”
    prompt: str

    #目标赋予prompt，如：“你的任务是......”
    query: str

    is_stream: bool

    context: list = []

    conversation_id: str

    def check_permission(self, tool: type[Tool] | Tool) -> bool:
        '''
        检查子agent的权限。
        tool 可以是 Tool 类或已实例化的 Tool 对象。
        '''
        # NOTE: 为了兼容 get_tools_lst() 现在返回实例对象，这里先判断是否是已实例化的 Tool。
        if isinstance(tool, Tool):
            tool_instance = tool
        else:
            tool_instance = tool()
        tool_perms = tool_instance.tool_permission_all()
        agent_perms = self.permission_all()
        return check_permission(agent_perms, tool_perms)    #No its not a recursion


    def get_available_tools(self) -> list:
        '''
        逐个过滤tools父类池，得到子类池，该类的工具权限：
        is_read [ ]
        is_write [ ]
        is_manage [ ]
        is_script [ ]
        is_destructive [ ]
        '''
        tools = get_tools_lst()
        available_tools = []
        for tool in tools:
            if self.check_permission(tool):
                # NOTE: get_tools_lst() 现已返回实例对象，无需再 tool() 构造。
                # 需要注入 subagent 的 Bashtools/SubAgentLauncher 已在 reAct.py 中按需重新实例化。
                available_tools.append(tool)
        return available_tools
    
    def has_manage_permission(self) -> bool:
        '''
        启动子agent的权限。
        '''
        return False

    def has_write_permission(self) -> bool:
        '''
        正常情况下，agent不允许写入文件，除非在子类中额外开放。
        '''
        return False
    
    def has_read_permission(self) -> bool:
        '''
        检查是否有读取文件的权限。
        '''
        return True
    
    def script_run_permission(self) -> bool:
        '''
        运行脚本的权限。
        '''
        return False
    
    def destructive_permission(self) -> bool:
        '''
        如rm，mkdir等对目录结构进行变化的操作。
        '''
        return False
    
    def permission_all(self) -> Permission:
        perms = Permission.NONE
        if self.has_read_permission():
            perms |= Permission.READ
        if self.has_write_permission():
            perms |= Permission.WRITE
        if self.has_manage_permission():
            perms |= Permission.MANAGE
        if self.script_run_permission():
            perms |= Permission.SCRIPT
        if self.destructive_permission():
            perms |= Permission.DESTRUCTIVE
        return perms
    
    def get_context(self) -> list:
        '''
        在后续子类中加载上下文。
        '''
        return []

    def _context_path(self) -> str:
        '''
        根据 conversation_id 返回上下文文件路径。
        '''
        from cloudra.config import get_config
        data_dir = get_config().data_dir
        if self.conversation_id and self.conversation_id != 'default':
            return os.path.join(data_dir, 'conversations', self.conversation_id, 'contexts', self.id + '.json')
        return os.path.join(data_dir, 'contexts', self.id + '.json')

    def _encode_image(self, image_path: str) -> str:
        """将本地图片转为 Base64"""
        import base64
        with open(image_path, "rb") as image_file:
            return base64.b64encode(image_file.read()).decode('utf-8')

    def llm_call(self) -> ChatResponse:
        '''
        调用llm接口。
        输出：
        1.若实例的is_stream为True，则返回ChatStream对象(一个文本块生成器)
        2.若实例的is_stream为False，则返回ChatResponse对象
        '''
        tool_classes = get_tools_lst()
        available_tool_classes = []
        for tool in tool_classes:
            if self.check_permission(tool):
                # HACK: 硬编码注入 subagent，使 Bashtools 的动态 description 生效
                if tool.tool_name == 'bashtools':
                    tool = tool.__class__(subagent=self)
                available_tool_classes.append(tool.tool_schema_json())
        
        client = OpenAI(api_key=self._api_key ,
                        base_url=self.llm_site ,
                        )

        def build_messages(use_multimodal=True):
            content = []
            content.append({"role": "system", "content": self.prompt})
            for context_block in self.context:
                msg = copy.deepcopy(dict(context_block))
                if 'role' not in msg: msg['role'] = 'user'
                
                # 处理多模态逻辑
                if isinstance(msg.get('content'), list):
                    new_content = []
                    for part in msg['content']:
                        if isinstance(part, dict) and part.get('type') == 'image_url':
                            path = part.get('image_url', {}).get('url')
                            if use_multimodal:
                                # 尝试构建 Base64
                                try:
                                    if path and os.path.exists(path):
                                        base64_image = self._encode_image(path)
                                        ext = os.path.splitext(path)[1].lower().replace('.', '')
                                        mime = f"image/{ext}" if ext in ['png', 'jpeg', 'jpg', 'webp'] else "image/png"
                                        new_content.append({
                                            "type": "image_url",
                                            "image_url": {"url": f"data:{mime};base64,{base64_image}"}
                                        })
                                    else:
                                        new_content.append({"type": "text", "text": f"[图片缺失: {path}]"})
                                except Exception as e:
                                    new_content.append({"type": "text", "text": f"[图片加载失败: {path}]"})
                            else:
                                # 降级模式：转为纯文本
                                new_content.append({"type": "text", "text": f"[图片: {path} (当前模型不支持视觉)]"})
                        else:
                            new_content.append(part)
                    msg['content'] = new_content if use_multimodal else "\n".join([p.get('text', '') for p in new_content])
                
                if 'content' not in msg:
                    msg['content'] = str(msg.get('assistant', msg)) if 'assistant' in msg else None
                
                if 'reasoning_content' in msg:
                    msg['reasoning_content'] = str(msg['reasoning_content']) if msg['reasoning_content'] else None
                
                content.append(msg)
            return content

        # 尝试第一次调用（完整多模态）
        try:
            messages = build_messages(use_multimodal=True)
            if available_tool_classes:
                return client.chat.completions.create(
                    model=self.model,
                    messages=messages,
                    temperature=self.temperature,
                    max_tokens=self.max_tokens,
                    top_p=self.top_p,
                    stream=self.is_stream,
                    tools=available_tool_classes,
                )
            else:
                return client.chat.completions.create(
                    model=self.model,
                    messages=messages,
                    temperature=self.temperature,
                    max_tokens=self.max_tokens,
                    top_p=self.top_p,
                    stream=self.is_stream
                )
        except Exception as e:
            # 捕获 400 错误（通常是由于不支持 image_url 引起的）
            err_msg = str(e).lower()
            if "image_url" in err_msg or "unknown variant" in err_msg or "400" in err_msg:
                print(f"\033[93m[系统预警]\033[0m 当前模型或接口不支持多模态内容，正在切换至“纯文本降级模式”重试...")
                # 重新构建纯文本消息列表
                messages = build_messages(use_multimodal=False)
                # 第二次调用（纯文本降级）
                if available_tool_classes:
                    return client.chat.completions.create(
                        model=self.model,
                        messages=messages,
                        temperature=self.temperature,
                        max_tokens=self.max_tokens,
                        top_p=self.top_p,
                        stream=self.is_stream,
                        tools=available_tool_classes,
                    )
                else:
                    return client.chat.completions.create(
                        model=self.model,
                        messages=messages,
                        temperature=self.temperature,
                        max_tokens=self.max_tokens,
                        top_p=self.top_p,
                        stream=self.is_stream
                    )
            else:
                # 其他类型的错误，原样抛出
                print(f"\033[91m[LLM 异常]\033[0m {e}")
                raise e
    
    def update_context(self, new_block: dict) -> None:
        '''
        更新上下文，新增一个块。
        确保每条消息都有 role 字段。
        文件路径根据 conversation_id 隔离。
        '''
        if 'role' not in new_block:
            new_block['role'] = 'user'
        
        # 提取 image_path（如果存在）
        image_path = new_block.pop('image_path', None)
        
        # 如果包含图片，将 content 构建为 OpenAI 标准的 list 结构，但 URL 暂存本地路径
        if image_path:
            text_content = new_block.get('content', "查看此图片内容：")
            new_block['content'] = [
                {"type": "text", "text": text_content if text_content else "查看此图片内容："},
                {
                    "type": "image_url", 
                    "image_url": {"url": image_path}
                }
            ]
        elif 'content' not in new_block:
            new_block['content'] = str(new_block)
                
        self.context.append(new_block)
        context_path = self._context_path()
        os.makedirs(os.path.dirname(context_path), exist_ok=True)
        with open(context_path, 'w', encoding='utf-8') as f:
            json.dump(self.context, f, ensure_ascii=False, indent=4)
    




class Coordinator(Agent):
    '''
    父agent，用于协调子agent。仅拥有管理工具。
    '''
    id: str = 'a-0'

    name: str = 'coordinator'

    type: str = 'coordinator'

    model: str = os.environ.get('COORDINATOR_MODEL')

    llm_site: str = os.environ.get('COORDINATOR_URL')

    _api_key: str = os.environ.get('COORDINATOR_KEY')

    temperature: float = 1.0

    max_tokens: int = 2000

    top_p: float = 1.0

    frequency_penalty: float = 0.0

    presence_penalty: float = 0.0

    prompt: str = '''
    你处在一个体系中，这个体系的名字叫做Cloudra。
    而你是一个多agent协作体系的协调者，你将直接对接用户的需求，将用户的需求拆解成一个个子任务，分发给子agent完成。在子agent发射后，你仍然拥有行动的能力，请用最理性实事求是的态度与用户说明情况。
    '''

    #由用户输入总任务。
    query: str

    is_stream: bool = True

    context: list = []

    def __init__(self, **data):
        super().__init__(**data)
        cwd = os.getcwd()
        self.prompt += f"\n\n当前项目的工作目录是: {cwd}\n在分发任务给子 Agent 时，请务必提供相关的绝对路径，确保它们能在正确的目录下操作。"
        self.context = self.get_context()

    def has_manage_permission(self) -> bool:
        '''
        协调者允许访问子agent。
        '''
        return True

    def get_context(self) -> list:
        '''
        加载上下文，但需要另行写入新增上下文
        '''
        system_json = {
            "role": "system",
            "content": self.prompt
        }
        query_json = {
            "role": "user",
            "content": self.query
        }
        context_path = self._context_path()
        os.makedirs(os.path.dirname(context_path), exist_ok=True)
        if os.path.isfile(context_path):
            try:
                with open(context_path, 'r', encoding='utf-8') as f:
                    self.context = json.load(f)
            except json.JSONDecodeError:
                print(f'无法解析JSON文件：{context_path}')
                exc_type, exc_value, exc_traceback = sys.exc_info()
                if exc_traceback is not None:
                    print(f'错误发生在文件 {exc_traceback.tb_frame.f_code.co_filename} 的第 {exc_traceback.tb_lineno} 行')
                    raise json.JSONDecodeError("无法解析JSON文件", context_path, exc_traceback.tb_lineno)
                else:
                    raise json.JSONDecodeError("无法解析JSON文件", context_path, 0)
        else:
            self.context = [system_json, query_json]
            with open(context_path, 'w', encoding='utf-8') as f:
                json.dump(self.context, f, ensure_ascii=False, indent=4)

        return self.context
    

        
class Follower(Agent):
    '''
    子agent大类，后续有细分类型，再定义子类。
    '''
    id: str

    name: str

    type: str = 'follower'

    model: str = os.environ.get('FOLLOWER_MODEL')

    llm_site: str = os.environ.get('FOLLOWER_URL')

    _api_key: str = os.environ.get('FOLLOWER_KEY')

    temperature: float = 1.0

    max_tokens: int = 2000

    top_p: float = 1.0

    frequency_penalty: float = 0.0

    presence_penalty: float = 0.0

    prompt: str

    query: str

    is_stream: bool = False

    context: list = []

    def __init__(self, **data):
        super().__init__(**data)
        self.context = self.get_context()

    def get_context(self) -> list:
        '''
        加载上下文，确保包含 prompt 和 query。
        '''
        system_json = {"role": "system", "content": self.prompt}
        query_json = {"role": "user", "content": self.query}
        
        context_path = self._context_path()
        os.makedirs(os.path.dirname(context_path), exist_ok=True)
        
        if os.path.isfile(context_path):
            try:
                with open(context_path, 'r', encoding='utf-8') as f:
                    history = json.load(f)
                    # 只有当历史为空或仅包含 system message 时才插入 query
                    if not history or (len(history) == 1 and history[0].get('role') == 'system'):
                        self.context = [system_json, query_json]
                    else:
                        self.context = history
            except:
                self.context = [system_json, query_json]
        else:
            self.context = [system_json, query_json]
        
        # 立即持久化，防止多进程环境下读取到空文件
        with open(context_path, 'w', encoding='utf-8') as f:
            json.dump(self.context, f, ensure_ascii=False, indent=4)
            
        return self.context

    def has_read_permission(self) -> bool:
        return True

def dynamic_deep_merge(buffer_dict: dict, delta_dict: dict) -> dict:
    """通用动态拼接工具：不硬编码任何字段名，自动合并字符串，自动递归字典"""
    for key, value in delta_dict.items():
        if isinstance(value, dict):
            # 如果是嵌套字典（比如 'function'），递归进去拼接
            if key not in buffer_dict:
                buffer_dict[key] = {}
            dynamic_deep_merge(buffer_dict[key], value)
            
        elif isinstance(value, str):
            # 如果是字符串（比如 'id', 'name', 'arguments'），执行累加操作
            if key not in buffer_dict:
                buffer_dict[key] = ""
            buffer_dict[key] += value
            
        else:
            # 如果是数字或其他类型（比如 'index'），直接覆盖/初始化
            buffer_dict[key] = value

    return buffer_dict

class Color:
    PURPLE = '\033[95m'
    CYAN = '\033[96m'
    DARKCYAN = '\033[36m'
    BLUE = '\033[94m'
    GREEN = '\033[92m'
    YELLOW = '\033[93m'
    RED = '\033[91m'
    BOLD = '\033[1m'
    UNDERLINE = '\033[4m'
    END = '\033[0m'

class reAct:
    session: Session
    agent: Agent
    prompt: str
    log: Log
    _continue : bool = True
    database_path: str = ''
    undos: list = []
    
    def __init__(self , agent: Agent, prompt: str = ""):
        self.agent = agent
        # 如果调用者显式提供了 prompt（即 Coordinator 定义的），则优先使用它实现任务隔离
        self.prompt = prompt if prompt else agent.prompt
        self.log = Log(agent.name)
        self.log.session = None
        
        data_dir = get_config().data_dir
        conv_id = getattr(agent, 'conversation_id', 'default')
        if conv_id and conv_id != 'default':
            log_path = os.path.join(data_dir, 'conversations', conv_id, 'logs', agent.name)
        else:
            log_path = os.path.join(data_dir, 'logs', agent.name)
        os.makedirs(os.path.dirname(log_path), exist_ok=True)

        self.log = Log(agent.name, log_path=log_path)
        self.log.session = None
        self.session_create()

    async def run(self):
        '''
        reAct的核心方法，执行agent的任务。
        '''
        while self._continue:
            self.session.state = 'thinking'
            tool_calls = self.reasoning()
            
            # 如果没有工具调用且有内容，或者已经停止
            if not tool_calls:
                # 检查最后一条消息内容
                content = self.agent.context[-1].get('content', '')
                if content:
                    self.session.state = 'sleeping'
                    # 更新数据库结果
                    con = sqlite3.connect(self.database_path, timeout=10)
                    cur = con.cursor()
                    cur.execute(
                        "UPDATE session SET state = ?, result = ? WHERE session_id = ?",
                        ('sleeping', content, self.session.id)
                    )
                    con.commit()
                    con.close()
                    return content
            
            self.session.state = 'acting'
            stopped = await self.act_result(tool_calls)
            
            if stopped:
                self.session.state = 'sleeping'
                content = self.agent.context[-1].get('content', '任务已完成')
                return content

    def reasoning(self) -> list:
        '''
        reAct的推理方法，分析agent的上下文和工具，决定下一步行动。
        '''
        agent = self.agent
        # 使用统一的 prompt
        agent.prompt = self.prompt
        response = agent.llm_call()
        final_response = ''
        reasoning_content = ''
        tool_calls = []
        message = None
        
        if agent.is_stream:
            tool_call_buffer = {}  
            self.log.update_content(f'{Color.CYAN}{agent.name}[reasoning]:{Color.END}\n')
            
            reasoning_buffer = ""
            
            for chunk in response:
                choice = chunk.choices[0]
                delta = choice.delta
                finish_reason = choice.finish_reason
                token = delta.content
                
                if hasattr(delta, 'reasoning_content') and delta.reasoning_content:
                    reasoning_buffer += delta.reasoning_content

                if token:
                    final_response += token
                    self.log.update_content(token)

                if delta.tool_calls:
                    for tc_chunk in delta.tool_calls:
                        tc_idx = tc_chunk.index
                        chunk_dict = tc_chunk.model_dump(exclude_none=True)

                        if tc_idx not in tool_call_buffer:
                            tool_call_buffer[tc_idx] = {}

                        tool_call_buffer[tc_idx] = dynamic_deep_merge(
                            tool_call_buffer[tc_idx], 
                            chunk_dict
                        )

                if finish_reason == 'tool_calls' or (finish_reason == 'stop' and tool_call_buffer):
                    if reasoning_buffer:
                        self.log.update_content(f"{Color.DARKCYAN}[thinking]{reasoning_buffer}{Color.END}\n")
                    reasoning_content = reasoning_buffer
                    
                    for tc_data in tool_call_buffer.values():
                        tool_calls.append(tc_data)
                        func_name = tc_data.get('function', {}).get('name', 'Unknown')
                        self.log.update_content(new_content=f"{Color.GREEN}工具调用: {func_name}{Color.END}\n")
            
            tool_call_buffer.clear()
             
        else:
            message = response.choices[0].message
            final_response = message.content
            if hasattr(message, 'reasoning_content') and message.reasoning_content:
                reasoning_content = message.reasoning_content
            self.log.update_content(f'{Color.CYAN}{agent.name}[reasoning]:{Color.END}\n{final_response}\n')
            
            if message.tool_calls:
                for tool_call in message.tool_calls:
                    tool_calls.append(tool_call.model_dump(exclude_none=True))

        # 构建完整的 assistant 消息写入上下文
        if message:  # 非stream模式
            new_block = message.model_dump(exclude_none=True)
        else:  # stream模式
            new_block = {'role': 'assistant'}
            if reasoning_content:
                new_block['reasoning_content'] = reasoning_content
            if final_response:
                new_block['content'] = final_response
            else:
                new_block['content'] = None
            
            if tool_calls:
                new_block['tool_calls'] = tool_calls
        
        # 延迟加载逻辑：如果有 tool_calls，先存入 undos 缓冲区
        if tool_calls:
            self.undos.append(new_block)
        else:
            agent.update_context(new_block)
            
        return tool_calls

    def _commit_undos(self):
        '''将缓冲区中的消息提交到上下文。'''
        while self.undos:
            self.agent.update_context(self.undos.pop(0))

    async def act_result(self, tool_calls: list):
        '''
        并发执行 reasoning 阶段收集到的 tool_calls。
        tool_calls 为标准 OpenAI 格式的 dict 列表。
        返回 bool：是否检测到 stop 工具并请求终止循环。
        '''
        session = self.session
        tools_list = get_tools_lst()
        
        # 并发执行所有工具调用
        acts = []
        tool_call_ids = []
        stop_requested = False
        
        for tool_call in tool_calls:
            tool_name = tool_call.get('function', {}).get('name', 'Unknown')
            tool_call_id = tool_call.get('id')
            tool_args_raw = tool_call.get('function', {}).get('arguments', {})
            
            try:
                if isinstance(tool_args_raw, str):
                    tool_args = json.loads(tool_args_raw)
                else:
                    tool_args = tool_args_raw
            except json.JSONDecodeError as e:
                # 如果 JSON 解析失败，将错误作为工具结果返回给 Agent
                result_block = {
                    'role': 'tool',
                    'tool_call_id': tool_call_id,
                    'content': f"【系统错误】工具参数解析失败 (JSON Error): {str(e)}。请确保参数格式正确。"
                }
                self._commit_undos()
                self.agent.update_context(result_block)
                self.log.update_content(f"{Color.RED}参数解析失败: {tool_name}{Color.END}\n")
                continue
            
            if tool_name == 'stop':
                self._continue = False
                stop_requested = True
            
            self.log.update_content(f"{Color.YELLOW}执行工具: {tool_name}{Color.END}\n")
            
            found = False
            for tool in tools_list:
                # NOTE: get_tools_lst() 现在返回实例对象（不再是类），因此直接用 tool.tool_name 匹配。
                if tool.tool_name == tool_name:
                    # 统一注入当前 agent 到工具实例中，以便工具内部获取上下文和进行安全审计
                    tool_instance = tool.__class__(subagent=self.agent)
                    
                    input_schema_class = tool_instance.input_schema
                    input_instance = input_schema_class(**tool_args)
                    session.state = 'waiting'
                    acts.append(tool_instance.safe_call(input_instance, session))
                    tool_call_ids.append(tool_call_id)
                    found = True
                    break
            
            if not found:
                result_block = {
                    'role': 'tool',
                    'tool_call_id': tool_call_id,
                    'content': f"【系统错误】找不到工具: {tool_name}。请检查工具名称是否正确。"
                }
                self._commit_undos()
                self.agent.update_context(result_block)
        
        if acts:
            # 使用 return_exceptions=True，确保单个工具崩溃不影响其他工具，且能捕获异常
            tool_results = await asyncio.gather(*acts, return_exceptions=True)
            self._commit_undos()
            
            # 每个工具结果单独作为一条消息
            for tool_result, tool_call_id in zip(tool_results, tool_call_ids):
                image_path = None
                if isinstance(tool_result, Exception):
                    # 如果工具执行抛出了异常，将错误信息反馈给 Agent
                    content = f"【系统错误】工具执行发生异常: {str(tool_result)}"
                    self.log.update_content(f"{Color.RED}工具异常: {str(tool_result)}{Color.END}\n")
                else:
                    result_dict = tool_result.toJson()
                    content = json.dumps(result_dict)
                    # 自动提取图片路径（如果工具输出中包含该字段）
                    if result_dict.get('image_path'):
                        image_path = result_dict.get('image_path')
                
                result_block = {
                    'role': 'tool',
                    'tool_call_id': tool_call_id,
                    'content': content
                }
                if image_path:
                    result_block['image_path'] = image_path
                    
                self.agent.update_context(result_block)
            
            session.state = 'thinking'

        return stop_requested

    def session_create(self) -> None:
        '''
        创建session，记录agent的整个生命周期。
        Session 文件放在 agent 所属 conversation 的 sessions/ 目录下。
        同时写入数据库作为单一事实来源。
        '''
        data_dir = get_config().data_dir
        if not self.database_path:
            self.database_path = os.path.join(data_dir, 'tasks', 'database.db')
        session_id = f'{self.agent.id}_?'
        conv_id = getattr(self.agent, 'conversation_id', 'default')
        if conv_id and conv_id != 'default':
            _path = os.path.join(data_dir, 'conversations', conv_id, 'sessions', session_id + '.json')
        else:
            _path = os.path.join(data_dir, 'sessions', session_id + '.json')
        os.makedirs(os.path.dirname(_path), exist_ok=True)
        session_ = Session(id=session_id, path=_path)
        session_.id = f'{self.agent.id}_{session_.create_time}'
        self.session = session_
        self.log.session = session_

        # 数据库为单一事实来源：创建即入库
        con = sqlite3.connect(self.database_path, timeout=10)
        cur = con.cursor()
        cur.execute("PRAGMA journal_mode=WAL;")
        cur.execute(
            "INSERT OR IGNORE INTO session (session_id, state, result) VALUES (?, ?, ?)",
            (session_.id, 'sleeping', '')
        )
        con.commit()
        con.close()
        return
    
    def kill(self):
        session = self.session
        session.state = 'killed'
        session.upload_json()
        self._continue = False
        # 同步数据库
        con = sqlite3.connect(self.database_path, timeout=10)
        cur = con.cursor()
        cur.execute(
            "UPDATE session SET state = ? WHERE session_id = ?",
            ('killed', session.id)
        )
        con.commit()
        con.close()

import time
import os

import logging

# 设置静默日志：仅打印关键错误
logging.basicConfig(
    level=logging.WARNING,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    filename='email_listener.log'
)
logger = logging.getLogger("EmailListener")

# 唤醒指令前缀
WAKEUP_PREFIX = "CLOUDRA:"

class EmailListener:
    def __init__(self, interval: int = 60):
        self.interval = interval
        self.config = get_config().email
        
        # 初始化读取器所需的 dummy agent
        from cloudra.agent import Coordinator
        # 避免监听器启动时产生冗余的 context 文件
        self.dummy_agent = Coordinator(query="listener_init", conversation_id="system_listener")
        self.reader = EmailReader(subagent=self.dummy_agent)

    def start(self):
        """静默启动监听循环"""
        if not self.config.user or not self.config.password:
            print("[错误] 邮件配置不完整，请确保 SMTP/IMAP 用户名和密码已设置。")
            return

        print(f"[*] Cloudra 邮件唤醒监听中... (前缀: {WAKEUP_PREFIX})")
        
        while True:
            try:
                self._poll_once()
            except Exception as e:
                logger.error(f"监听循环异常: {e}")
            
            time.sleep(self.interval)

    def _poll_once(self):
        # 仅读取 1 封最新的未读邮件
        # IMAP 的 _read_imap 方法在读取后会自动将邮件设为已读
        emails = self.reader._read_imap(self.config, EmailReader_input_schema(count=1, only_unread=True))
        
        if not emails:
            return

        msg = emails[0]
        
        # 白名单校验
        if self.config.whitelist:
            # 提取发件人邮箱 (处理 "Name <email@example.com>" 格式)
            import re
            sender_email = msg.sender
            match = re.search(r'<(.*)>', sender_email)
            if match:
                sender_email = match.group(1)
            
            if sender_email not in self.config.whitelist:
                logger.warning(f"拦截到非法发件人的唤醒指令: {sender_email}")
                return

        subject = msg.subject.strip()
        body = msg.body.strip()

        # 检测唤醒指令：优先查主题，其次查正文第一行
        query = None
        if subject.upper().startswith(WAKEUP_PREFIX):
            query = subject[len(WAKEUP_PREFIX):].strip()
        elif body.upper().startswith(WAKEUP_PREFIX):
            # 提取第一行作为指令
            first_line = body.split('\n')[0]
            query = first_line[len(WAKEUP_PREFIX):].strip()

        if query:
            print(f"\n[!] 唤醒指令触发: {query} (来自: {msg.sender})")
            
            # 记录到日志
            logger.warning(f"触发任务: {query} | 来源: {msg.sender}")

            try:
                # 开启任务执行
                # 注意：此处为同步执行，执行期间监听会暂停，防止并发冲突
                run_task(query, verbose=True)
            except Exception as e:
                logger.error(f"执行唤醒任务失败: {e}")
                print(f"[错误] 任务执行中断: {e}")

if __name__ == "__main__":
    # 可以通过环境变量自定义轮询间隔
    interval = int(os.environ.get("CLOUDRA_POLL_INTERVAL", 60))
    listener = EmailListener(interval=interval)
    listener.start()
